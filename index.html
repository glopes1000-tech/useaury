let categories = []; // Categorias Padrão const defaultCategories = [ { id: 'aneis', name: 'Anéis' }, { id: 'gargantilhas', name: 'Gargantilhas' }, { id: 'brincos', name: 'Brincos' }, { id: 'pulseiras', name: 'Pulseiras' }, { id: 'escapularios', name: 'Escapulários' }, { id: 'conjuntos', name: 'Conjuntos' } ]; // Carregar categorias do Firestore window.loadCategoriesFromFirebase = async function() { try { const querySnapshot = await getDocs(collection(db, "categories")); categories = []; querySnapshot.forEach((docSnap) => { categories.push({ id: docSnap.id, ...docSnap.data() }); }); if (categories.length === 0) { for (const cat of defaultCategories) { await setDoc(doc(db, "categories", cat.id), { name: cat.name }); } categories = [...defaultCategories]; } renderCategoryUI(); } catch (error) { console.error("Erro ao carregar categorias:", error); categories = [...defaultCategories]; renderCategoryUI(); } }; // Cadastrar nova categoria window.saveCategoryToFirebase = async function(event) { event.preventDefault(); const catNameInput = document.getElementById('cat-name'); const name = catNameInput.value.trim(); if (!name) return; const id = name.toLowerCase() .normalize('NFD').replace(/[\u0300-\u036f]/g, '') .replace(/[^a-z0-9]/g, '-'); try { await setDoc(doc(db, "categories", id), { name }); alert(`Categoria "${name}" cadastrada com sucesso!`); catNameInput.value = ''; await loadCategoriesFromFirebase(); renderAdminCategoryList(); } catch (error) { console.error("Erro ao salvar categoria:", error); alert("Erro ao cadastrar categoria."); } }; // Excluir categoria window.deleteCategoryFromFirebase = async function(id) { if (confirm("Tem certeza que deseja excluir esta categoria?")) { try { await deleteDoc(doc(db, "categories", id)); alert("Categoria excluída com sucesso!"); await loadCategoriesFromFirebase(); renderAdminCategoryList(); } catch (error) { console.error("Erro ao excluir categoria:", error); } } }; // Atualizar interfaces de categoria window.renderCategoryUI = function() { const navMenu = document.getElementById('nav-categories'); if (navMenu) { navMenu.innerHTML = ``; categories.forEach(cat => { navMenu.innerHTML += ``; }); } const filterContainer = document.getElementById('category-filters'); if (filterContainer) { filterContainer.innerHTML = ``; categories.forEach(cat => { filterContainer.innerHTML += ``; }); } const selectCat = document.getElementById('prod-cat'); if (selectCat) { selectCat.innerHTML = ''; categories.forEach(cat => { selectCat.innerHTML += ``; }); } }; window.renderAdminCategoryList = function() { const list = document.getElementById('admin-categories-list'); if (!list) return; list.innerHTML = ''; categories.forEach(c => { list.innerHTML += `
${c.name}
`; }); };
